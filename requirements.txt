# Pinned to the exact versions this build was developed and tested against
# (captured 2026-08). Unpinned requirements meant a fresh `pip install` on a
# new machine could silently grab a newer, untested release of any of these
# — including ones with breaking API changes (the Google/MSAL/Groq SDKs move
# fast) — and fail in ways that have nothing to do with this app's own code.
# To intentionally upgrade one package, bump it here and re-test the
# feature(s) it touches rather than just deleting the pin.

# --- Timezone-aware "current time" for the LLM (app.py's LOCAL_TZ) --------
# stdlib zoneinfo needs an IANA tz database; not every base Linux image
# ships one. This pip package is zoneinfo's documented fallback source.
tzdata==2025.2

# --- TLS trust (see cert_bootstrap.py) ------------------------------------
# Pinned explicitly (not just relied on as a transitive dep of requests/
# edge-tts) because cert_bootstrap.py reaches into certifi's internals
# (certifi.where / certifi.core.where) at import time — a certifi release
# that renames/removes those would break cert_bootstrap.py silently.
certifi==2025.4.26
